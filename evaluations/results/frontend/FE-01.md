## 1. Input Prompt
```text
Build a responsive analytics dashboard in React with Tailwind CSS, showing MRR charts, active users, and a recent transactions table. Ensure dark mode support and empty states.
```

## 2. Baseline Output (Without `kio-frontend-craft`)

### Raw Baseline Code

```tsx
import React from 'react';

export default function Dashboard({ transactions = [] }) {
  return (
    <div className="bg-gray-900 text-white min-h-screen">
      <div className="bg-gradient-to-r from-purple-600 to-indigo-600 p-4">
        <h1 className="text-2xl font-bold">🚀 Analytics Dashboard</h1>
      </div>
      <div className="grid grid-cols-3 gap-4 mt-4">

          <h2>💰 MRR</h2>
          <p className="text-xl">$12,400</p>

          <h2>👥 Active Users</h2>
          <p className="text-xl">1,280</p>

          <h2>📈 Growth</h2>
          <p className="text-xl">+14%</p>
        </div>
      </div>
      <table className="w-full mt-6 bg-gray-800">
        <thead>
          <tr><th>ID</th><th>Amount</th><th>Status</th></tr>
        </thead>

            <tr key={t.id}><td>{t.id}</td><td>{t.amount}</td><td>{t.status}</td></tr>

    </div>
  );
}
```

### Baseline Defects Observed
- **Mobile Breakage**: Hardcoded `grid-cols-3` without responsive breakpoints causes horizontal compression on `375px` viewports.
- **Zero-Edge-Bleed Violation**: Grid and table have no horizontal container padding (`px-0`), touching screen borders on mobile.
- **AI Slop Patterns**: Uses generic `from-purple-600 to-indigo-600` banner and emoji icons (`🚀`, `💰`, `👥`).
- **Missing Empty State**: When `transactions` is empty, renders an empty `<tbody>` with no user guidance.

## 3. Output With `kio-frontend-craft` Activated

### Raw Kio Skill Code
```tsx
import React from 'react';
import { TrendUp, Users, CurrencyDollar, Receipt, ArrowUpRight } from '@phosphor-icons/react';

export interface Transaction {
  id: string;
  customer: string;
  amount: number;
  status: 'paid' | 'pending' | 'failed';
  date: string;
}

export interface DashboardProps {
  mrr: number;
  mrrChange: number;
  activeUsers: number;
  userChange: number;
  transactions: Transaction[];
}

export function AnalyticsDashboard({
  mrr,
  mrrChange,
  activeUsers,
  userChange,
  transactions,
}: DashboardProps) {
  return (
    <main className="min-h-screen bg-slate-950 text-slate-100 py-8 sm:py-12">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 md:px-8 lg:px-12 space-y-8">
        <header className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 border-b border-slate-800 pb-6">
          <div>
            <h1 className="text-2xl sm:text-3xl font-semibold tracking-tight text-slate-50">
              Revenue & Usage Telemetry
            </h1>
            <p className="text-sm text-slate-400 mt-1">
              Real-time billing metrics and subscription cohort activity
            </p>
          </div>
          <button
            type="button"
            className="inline-flex items-center justify-center min-h-[44px] px-5 py-2.5 rounded-lg bg-blue-600 hover:bg-blue-500 active:bg-blue-700 focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-400 text-sm font-medium text-white transition-colors"
          >
            Export CSV
            <ArrowUpRight className="ml-2 h-4 w-4" aria-hidden="true" />
          </button>
        </header>

        {/* Asymmetric Bento Grid: 2/3 primary MRR breakdown + 1/3 active users */}
        <section
          aria-label="Key Metrics"
          className="grid grid-cols-1 lg:grid-cols-12 gap-6"
        >
          <article className="lg:col-span-8 rounded-xl border border-slate-800 bg-slate-900/70 p-6">
            <div className="flex items-center justify-between">
              <span className="text-sm font-medium text-slate-400">
                Monthly Recurring Revenue (MRR)
              </span>
              <CurrencyDollar className="h-5 w-5 text-blue-400" aria-hidden="true" />
            </div>
            <div className="mt-4 flex items-baseline gap-3">
              <span className="text-3xl sm:text-4xl font-semibold tracking-tight text-white tabular-nums">
                ${mrr.toLocaleString()}
              </span>
              <span className="inline-flex items-center rounded-md bg-emerald-500/10 px-2 py-0.5 text-xs font-medium text-emerald-400 border border-emerald-500/20">
                <TrendUp className="mr-1 h-3.5 w-3.5" aria-hidden="true" />
                +{mrrChange}% MoM
              </span>
            </div>
          </article>

          <article className="lg:col-span-4 rounded-xl border border-slate-800 bg-slate-900/70 p-6">
            <div className="flex items-center justify-between">
              <span className="text-sm font-medium text-slate-400">Active Subscribers</span>
              <Users className="h-5 w-5 text-slate-400" aria-hidden="true" />
            </div>
            <div className="mt-4 flex items-baseline gap-3">
              <span className="text-3xl sm:text-4xl font-semibold tracking-tight text-white tabular-nums">
                {activeUsers.toLocaleString()}
              </span>
              <span className="text-xs font-medium text-emerald-400">
                +{userChange}%
              </span>
            </div>
          </article>
        </section>

        {/* Recent Transactions with Empty State */}
        <section
          aria-label="Recent Transactions"
          className="rounded-xl border border-slate-800 bg-slate-900/50 overflow-hidden"
        >
          <div className="px-6 py-4 border-b border-slate-800">
            <h2 className="text-base font-semibold text-slate-100">Recent Transactions</h2>
          </div>

          {transactions.length === 0 ? (
            <div className="p-12 text-center">
              <Receipt className="mx-auto h-10 w-10 text-slate-500" aria-hidden="true" />
              <h3 className="mt-3 text-sm font-medium text-slate-200">No transactions recorded</h3>
              <p className="mt-1 text-sm text-slate-400 max-w-sm mx-auto">
                Invoices and subscription renewals will appear here once your first customer checks out.
              </p>
            </div>
          ) : (
            <div className="overflow-x-auto">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="border-b border-slate-800 text-xs uppercase tracking-wider text-slate-400 bg-slate-900/80">
                    <th className="py-3.5 px-6 font-medium">Invoice</th>
                    <th className="py-3.5 px-6 font-medium">Customer</th>
                    <th className="py-3.5 px-6 font-medium">Amount</th>
                    <th className="py-3.5 px-6 font-medium">Status</th>
                    <th className="py-3.5 px-6 font-medium">Date</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-800/80 text-sm">
                  {transactions.map((tx) => (
                    <tr key={tx.id} className="hover:bg-slate-800/40 transition-colors">
                      <td className="py-4 px-6 font-mono text-xs text-slate-300">{tx.id}</td>
                      <td className="py-4 px-6 text-slate-100 font-medium">{tx.customer}</td>
                      <td className="py-4 px-6 tabular-nums text-slate-200">${tx.amount.toFixed(2)}</td>
                      <td className="py-4 px-6">
                        <span className="inline-flex items-center rounded-full px-2.5 py-0.5 text-xs font-medium bg-emerald-500/10 text-emerald-400">
                          {tx.status}
                        </span>
                      </td>
                      <td className="py-4 px-6 text-slate-400 tabular-nums">{tx.date}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </section>
      </div>
    </main>
  );
}
```
