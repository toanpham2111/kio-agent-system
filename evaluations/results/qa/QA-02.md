## 1. Input Prompt
```text
Write a Vitest unit test suite for a CartCheckoutService that reserves inventory via an external InventoryClient and charges the customer via a StripeGateway.
```

---

## 2. Baseline Output (Without `kio-qa-security`)

### Raw Baseline Code
```typescript
import { describe, it, expect, vi } from 'vitest';
import { CartCheckoutService } from './CartCheckoutService';

describe('CartCheckoutService', () => {
  it('processes checkout successfully', async () => {
    const service = new CartCheckoutService();
    // Anti-pattern: Mocking the unit under test!
    vi.spyOn(service, 'checkout').mockResolvedValue({ status: 'success', orderId: '123' });

    const res = await service.checkout({ cartId: 'c1', amount: 5000 });
    expect(res.status).toBe('success');
  });
});
```

### Baseline Defects Observed
- **Mocking the Unit Under Test (Tautological Test)**: `vi.spyOn(service, 'checkout').mockResolvedValue(...)` replaces the method being tested with a mock and asserts that the mock returns what it was told to return (0% real logic coverage).
- **Zero Failure-Mode or Rollback Verification**: Does not test what happens when `StripeGateway.charge` fails after `InventoryClient.reserve` succeeds.

## 3. Output With `kio-qa-security` Activated

### Raw Kio Skill Code
```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { CartCheckoutService, InventoryClient, PaymentGateway } from './CartCheckoutService';

describe('CartCheckoutService (Disciplined Mocking)', () => {
  let inventoryClient: InventoryClient;
  let paymentGateway: PaymentGateway;
  let checkoutService: CartCheckoutService;

  beforeEach(() => {
    // Mock ONLY external I/O boundaries; execute 100% real CartCheckoutService logic
    inventoryClient = {
      reserve: vi.fn(),
      release: vi.fn(),
    };
    paymentGateway = {
      charge: vi.fn(),
    };
    checkoutService = new CartCheckoutService(inventoryClient, paymentGateway);
  });

  it('completes checkout and forwards deterministic idempotency key', async () => {
    vi.mocked(inventoryClient.reserve).mockResolvedValue({ reservationId: 'res_01' });
    vi.mocked(paymentGateway.charge).mockResolvedValue({ transactionId: 'tx_99', status: 'succeeded' });

    const result = await checkoutService.checkout({
      cartId: 'cart_abc',
      userId: 'usr_1',
      items: [{ sku: 'SKU-1', qty: 2, priceCents: 2500 }],
    });

    expect(result.status).toBe('completed');
    expect(result.totalCents).toBe(5000);
    expect(paymentGateway.charge).toHaveBeenCalledWith({
      amountCents: 5000,
      userId: 'usr_1',
      idempotencyKey: 'checkout:cart_abc',
    });
    expect(inventoryClient.release).not.toHaveBeenCalled();
  });

  it('releases reserved inventory when payment gateway declines charge', async () => {
    vi.mocked(inventoryClient.reserve).mockResolvedValue({ reservationId: 'res_02' });
    vi.mocked(paymentGateway.charge).mockRejectedValue(new Error('CARD_DECLINED'));

    await expect(
      checkoutService.checkout({
        cartId: 'cart_fail',
        userId: 'usr_1',
        items: [{ sku: 'SKU-1', qty: 1, priceCents: 2500 }],
      })
    ).rejects.toThrow('CARD_DECLINED');

    // Compensating transaction must release stock
    expect(inventoryClient.release).toHaveBeenCalledTimes(1);
    expect(inventoryClient.release).toHaveBeenCalledWith('res_02');
  });

  it('aborts before calling payment gateway if inventory reservation fails', async () => {
    vi.mocked(inventoryClient.reserve).mockRejectedValue(new Error('OUT_OF_STOCK'));

    await expect(
      checkoutService.checkout({
        cartId: 'cart_oos',
        userId: 'usr_1',
        items: [{ sku: 'SKU-1', qty: 5, priceCents: 1000 }],
      })
    ).rejects.toThrow('OUT_OF_STOCK');

    expect(paymentGateway.charge).not.toHaveBeenCalled();
  });
});
```
