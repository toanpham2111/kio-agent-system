# syntax=docker/dockerfile:1.7

ARG NODE_VERSION=20.18.1
ARG ALPINE_VERSION=3.20

# ---------- Base ----------
FROM node:${NODE_VERSION}-alpine${ALPINE_VERSION} AS base
# libc6-compat is needed by some native Next.js/SWC binaries on Alpine
RUN apk add --no-cache libc6-compat
WORKDIR /app

# ---------- Dependencies ----------
FROM base AS deps
COPY package.json package-lock.json* yarn.lock* pnpm-lock.yaml* ./
RUN --mount=type=cache,target=/root/.npm \
    --mount=type=cache,target=/usr/local/share/.cache \
    --mount=type=cache,target=/root/.local/share/pnpm/store \
    if [ -f package-lock.json ]; then npm ci; \
    elif [ -f yarn.lock ]; then yarn install --frozen-lockfile; \
    elif [ -f pnpm-lock.yaml ]; then corepack enable pnpm && pnpm install --frozen-lockfile; \
    else echo "No lockfile found." && exit 1; \
    fi

# ---------- Build ----------
FROM base AS builder
ENV NEXT_TELEMETRY_DISABLED=1 \
    NODE_ENV=production
COPY --from=deps /app/node_modules ./node_modules
COPY . .
# NEXT_PUBLIC_* vars are inlined at build time; pass them as build args if needed:
# ARG NEXT_PUBLIC_API_URL
# ENV NEXT_PUBLIC_API_URL=$NEXT_PUBLIC_API_URL
RUN if [ -f package-lock.json ]; then npm run build; \
    elif [ -f yarn.lock ]; then yarn build; \
    elif [ -f pnpm-lock.yaml ]; then corepack enable pnpm && pnpm build; \
    fi

# ---------- Runtime ----------
FROM node:${NODE_VERSION}-alpine${ALPINE_VERSION} AS runner
WORKDIR /app

ENV NODE_ENV=production \
    NEXT_TELEMETRY_DISABLED=1 \
    PORT=3000 \
    HOSTNAME=0.0.0.0

# Non-root user with fixed UID/GID (friendly to Kubernetes runAsNonRoot)
RUN addgroup -g 1001 -S nodejs && adduser -S nextjs -u 1001 -G nodejs

# Static assets and public files
COPY --from=builder --chown=nextjs:nodejs /app/public ./public

# Standalone server (includes the traced minimal node_modules)
COPY --from=builder --chown=nextjs:nodejs /app/.next/standalone ./
COPY --from=builder --chown=nextjs:nodejs /app/.next/static ./.next/static

# Writable dir for ISR/image-optimization cache
RUN mkdir -p .next/cache && chown -R nextjs:nodejs .next

USER 1001

EXPOSE 3000

HEALTHCHECK --interval=30s --timeout=5s --start-period=15s --retries=3 \
    CMD node -e "fetch('http://127.0.0.1:'+process.env.PORT+'/').then(r=>process.exit(r.status<500?0:1)).catch(()=>process.exit(1))"

# Use node directly (not npm) so SIGTERM reaches the process
CMD ["node", "server.js"]

**Notes**

- **`public/` must exist** in the repo, or the `COPY` fails. Add an empty `public/.gitkeep` if you have none.
- **Monorepos:** standalone output nests under the workspace path, for example `.next/standalone/apps/web/server.js`. Adjust the `COPY` paths and `CMD` accordingly.
- **Runtime env vars** (secrets, DB URLs) are supplied at `docker run` or in your orchestrator. `.env*` is excluded from the image on purpose.
- **Health check:** if `/` needs auth or hits a database, point it at a dedicated `/api/health` route.
- **Read-only root filesystem:** mount a writable volume or tmpfs at `/app/.next/cache`.
- **Sharp:** Next 14 standalone auto-includes `sharp` for image optimization when it's installed. Add it to your dependencies if you use `next/image`.
- **Build and run:**
  ```bash
  docker build -t myapp .
  docker run -p 3000:3000 --env-file .env.production myapp
  ```

## `Dockerfile`

```dockerfile
# syntax=docker/dockerfile:1.7

# Pin to a minor version. For production deploys, reference the digest:
#   node:22-bookworm-slim@sha256:<digest>
ARG NODE_VERSION=22

# ---------- 1. deps: install dependencies only ----------
FROM node:${NODE_VERSION}-bookworm-slim AS deps
WORKDIR /app
COPY package.json package-lock.json ./
RUN --mount=type=cache,target=/root/.npm \
    npm ci

# ---------- 2. build: compile Next.js (output: 'standalone') ----------
FROM node:${NODE_VERSION}-bookworm-slim AS build
WORKDIR /app
ENV NEXT_TELEMETRY_DISABLED=1
COPY --from=deps /app/node_modules ./node_modules
COPY . .
# Public build-time values (NEXT_PUBLIC_*) go through ARG. Never put secrets in ARG/ENV.
# For a secret needed at build time, use: RUN --mount=type=secret,id=...
RUN npm run build

# ---------- 3. runner: minimal, non-root ----------
FROM node:${NODE_VERSION}-bookworm-slim AS runner
WORKDIR /app

ENV NODE_ENV=production \
    NEXT_TELEMETRY_DISABLED=1 \
    PORT=3000 \
    HOSTNAME=0.0.0.0

# The standalone bundle already contains only the node_modules that are used.
COPY --from=build --chown=node:node /app/.next/standalone ./
COPY --from=build --chown=node:node /app/.next/static ./.next/static
COPY --from=build --chown=node:node /app/public ./public

# Writable dir for the ISR/image cache. This lets the container run with --read-only.
RUN mkdir -p .next/cache && chown node:node .next/cache

# The built-in `node` user is UID 1000, non-root. Use numeric UID 10001 instead if your orchestrator
# enforces runAsNonRoot with a UID >= 10000.
USER node

EXPOSE 3000

HEALTHCHECK --interval=30s --timeout=3s --start-period=15s --retries=3 \
  CMD node -e "fetch('http://127.0.0.1:3000/api/health').then(r=>process.exit(r.ok?0:1)).catch(()=>process.exit(1))"

# `node` runs as PID 1 and receives SIGTERM directly. Next's standalone server shuts down gracefully.
# Use `docker run --init` or tini if your code spawns child processes.
CMD ["node", "server.js"]
```

## Notes

- **Package manager:** the file assumes npm with `package-lock.json`. For pnpm or yarn, swap the `deps` stage for `corepack enable && pnpm install --frozen-lockfile`.
- **Runtime secrets:** pass them as environment variables or from a secret store. `.env*` is excluded from the build context.
- **Monorepo:** if the project is a monorepo or sets `outputFileTracingRoot`, the standalone folder nests the app path. In that case, copy `.next/standalone` and run `node apps/web/server.js`.
- **`sharp`:** Next 14 uses it for `next/image` optimization. If you run `npm ci` on glibc (bookworm), the prebuilt binary is included automatically.
- **CI:** scan the image with Trivy or Grype and fail on fixable CRITICAL or HIGH findings. Rebuild weekly to pick up base-image patches.
