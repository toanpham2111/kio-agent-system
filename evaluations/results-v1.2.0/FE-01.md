## Notes

- **Dark mode:** it uses Tailwind's `class` strategy. The initial value comes from `localStorage` or `prefers-color-scheme`, and the toggle persists it. Every surface, border, text and chart colour has an explicit `dark:` variant. SVG strokes and fills use `currentColor` or `fill-*` classes, so charts follow the theme.
- **Empty states:** each widget handles `[]` on its own. The KPI cards show "—" when there's no data. The header's "Show empty" button previews them. Remove it in production.
- **Responsive layout:**
  - KPIs go from 2 columns to 4.
  - The charts go from stacked to a 2/3 + 1/3 split at `lg`.
  - The SVG charts scale through `viewBox`.
  - Transactions render as a table at `md` and up, and as compact cards below that.
- **Accessibility:** the chart has `role="img"` with a label, and its hover zones are focusable. The table uses `<th scope>`. The toggles use `aria-pressed`. Status is shown as text, not colour alone.
- **Real data:** pass `data={{ mrr, activeUsers, transactions }}` into `<App />`. For loading and error states, add a skeleton or error branch in the same places as `EmptyState`.

function niceMax(v: number) {
  if (v <= 0) return 1;

function useElementWidth<T extends HTMLElement>() {
  const ref = useRef<T>(null);
  const [w, setW] = useState(0);
  useLayoutEffect(() => {
    const el = ref.current;
    if (!el) return;
    const ro = new ResizeObserver(([e]) => setW(Math.floor(e.contentRect.width)));
    ro.observe(el);
    setW(el.clientWidth);
    return () => ro.disconnect();
  }, []);
  return [ref, w] as const;
}

  title, subtitle, action, children, className = "",
}: {
  title: string; subtitle?: string; action?: React.ReactNode;
  children: React.ReactNode; className?: string;
}) {
  return (
    <section
      aria-label={title}
      className={`rounded-xl border border-slate-200 bg-white p-4 sm:p-6 shadow-sm
                  dark:border-slate-800 dark:bg-slate-900 ${className}`}
    >
      <header className="mb-4 flex items-start justify-between gap-3">
        <div className="min-w-0">
          <h2 className="text-base font-semibold text-slate-900 dark:text-slate-50">{title}</h2>
          {subtitle && <p className="mt-0.5 text-sm text-slate-600 dark:text-slate-400">{subtitle}</p>}
        </div>
        {action}
      </header>
      {children}
    </section>
  );
}

function EmptyState({
  icon: Icon, title, description, actionLabel, onAction,
}: {
  icon: React.ComponentType<{ className?: string; "aria-hidden"?: boolean }>;
  title: string; description: string; actionLabel?: string; onAction?: () => void;
}) {
  return (
    <div className="flex min-h-56 flex-col items-center justify-center rounded-lg border border-dashed
                    border-slate-300 px-4 py-10 text-center dark:border-slate-700">
      <span className="mb-3 grid size-11 place-items-center rounded-full bg-slate-100 dark:bg-slate-800">
        <Icon className="size-5 text-slate-600 dark:text-slate-400" aria-hidden />
      </span>
      <h3 className="text-sm font-semibold text-slate-900 dark:text-slate-50">{title}</h3>
      <p className="mt-1 max-w-xs text-sm text-slate-600 dark:text-slate-400">{description}</p>
      {actionLabel && onAction && (
        <button
          onClick={onAction}
          className="mt-4 inline-flex min-h-11 items-center gap-2 rounded-lg bg-teal-700 px-5 py-2.5 text-sm

function Skeleton({ className = "" }: { className?: string }) {
  return <div className={`motion-safe:animate-pulse rounded-md bg-slate-200 dark:bg-slate-800 ${className}`} />;
}

/* ───────────── KPI cards ───────────── */
function Kpi({
  label, value, delta, icon: Icon, loading,
}: {
  label: string; value: string | null; delta: number | null;
  icon: React.ComponentType<{ className?: string; "aria-hidden"?: boolean }>; loading?: boolean;
}) {
  const up = (delta ?? 0) >= 0;
  const Trend = up ? TrendingUp : TrendingDown;
  return (
    <div className="rounded-xl border border-slate-200 bg-white p-4 sm:p-6 shadow-sm dark:border-slate-800 dark:bg-slate-900">
      <div className="flex items-center justify-between">
        <p className="text-sm font-medium text-slate-600 dark:text-slate-400">{label}</p>
        <Icon className="size-5 text-slate-500 dark:text-slate-400" aria-hidden />
      </div>
      {loading ? (
        <Skeleton className="mt-3 h-8 w-28" />
      ) : (
        <p className="mt-2 text-2xl font-semibold tabular-nums text-slate-900 dark:text-slate-50 sm:text-3xl">
          {value ?? "—"}
        </p>
      )}
      <p className="mt-2 flex items-center gap-1.5 text-sm">
        {delta === null || loading ? (
          <span className="text-slate-600 dark:text-slate-400">No comparison yet</span>
        ) : (
          <>
            <Trend
              className={`size-4 ${up ? "text-emerald-700 dark:text-emerald-400" : "text-red-700 dark:text-red-400"}`}
              aria-hidden
            />
            <span className={`font-medium ${up ? "text-emerald-700 dark:text-emerald-400" : "text-red-700 dark:text-red-400"}`}>
              {pct(delta)}
            </span>
            <span className="text-slate-600 dark:text-slate-400">vs last period</span>
          </>
        )}
      </p>
    </div>
  );
}

/* ───────────── MRR line/area chart ───────────── */
function MrrChart({ data }: { data: MrrPoint[] }) {
  const [wrap, width] = useElementWidth<HTMLDivElement>();
  const [active, setActive] = useState<number | null>(null);
  const uid = useRef(`g${Math.random().toString(36).slice(2, 8)}`).current;

  const H = 260;
  const m = { t: 12, r: 12, b: 28, l: width < 480 ? 44 : 56 };
  const iw = Math.max(width - m.l - m.r, 0);
  const ih = H - m.t - m.b;
  const max = niceMax(Math.max(...data.map((d) => d.mrr)));
  const x = (i: number) => m.l + (data.length === 1 ? iw / 2 : (i / (data.length - 1)) * iw);
  const y = (v: number) => m.t + ih - (v / max) * ih;

  const line = data.map((d, i) => `${i ? "L" : "M"}${x(i).toFixed(1)},${y(d.mrr).toFixed(1)}`).join(" ");
  const area = `${line} L${x(data.length - 1)},${m.t + ih} L${x(0)},${m.t + ih} Z`;
  const ticks = [0, 0.25, 0.5, 0.75, 1].map((t) => t * max);
  const labelEvery = width < 480 ? Math.ceil(data.length / 4) : width < 768 ? 2 : 1;

  const onMove = (e: React.PointerEvent<SVGSVGElement>) => {
    const r = e.currentTarget.getBoundingClientRect();
    const px = e.clientX - r.left;
    const i = Math.round(((px - m.l) / (iw || 1)) * (data.length - 1));
    setActive(Math.min(Math.max(i, 0), data.length - 1));
  };
  const onKey = (e: React.KeyboardEvent) => {
    if (e.key === "ArrowRight") setActive((a) => Math.min((a ?? -1) + 1, data.length - 1));
    else if (e.key === "ArrowLeft") setActive((a) => Math.max((a ?? data.length) - 1, 0));
    else if (e.key === "Escape") setActive(null);
    else return;
    e.preventDefault();
  };

  const a = active !== null ? data[active] : null;

  return (
    <div ref={wrap} className="relative w-full">
      {width > 0 && (
        <svg
          width={width}
          height={H}
          role="img"
          tabIndex={0}
          aria-label={`Monthly recurring revenue, ${data[0].month} to ${data.at(-1)!.month}. Use left and right arrow keys to inspect each month.`}
          onPointerMove={onMove}
          onPointerLeave={() => setActive(null)}
          onKeyDown={onKey}
          onBlur={() => setActive(null)}
          className="block touch-pan-y rounded-md focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-teal-600"
        >
          <defs>
            <linearGradient id={uid} x1="0" y1="0" x2="0" y2="1">
              <stop offset="0%" className="[stop-color:theme(colors.teal.600)] dark:[stop-color:theme(colors.teal.400)]" stopOpacity="0.22" />
              <stop offset="100%" className="[stop-color:theme(colors.teal.600)] dark:[stop-color:theme(colors.teal.400)]" stopOpacity="0" />
            </linearGradient>
          </defs>

          {ticks.map((t) => (
            <g key={t}>
              <line x1={m.l} x2={width - m.r} y1={y(t)} y2={y(t)} className="stroke-slate-200 dark:stroke-slate-800" />
              <text x={m.l - 8} y={y(t)} dy="0.32em" textAnchor="end" className="fill-slate-600 text-[11px] dark:fill-slate-400">
                {usd(t, true)}
              </text>
            </g>
          ))}

          {data.map((d, i) =>
            i % labelEvery === 0 ? (
              <text key={d.month} x={x(i)} y={H - 8} textAnchor="middle" className="fill-slate-600 text-[11px] dark:fill-slate-400">
                {d.month}
              </text>
            ) : null,
          )}

          <path d={area} fill={`url(#${uid})`} />
          <path d={line} fill="none" strokeWidth={2} strokeLinejoin="round" strokeLinecap="round"
                className="stroke-teal-700 dark:stroke-teal-400" />

          {a && active !== null && (
            <g>
              <line x1={x(active)} x2={x(active)} y1={m.t} y2={m.t + ih} className="stroke-slate-300 dark:stroke-slate-600" strokeDasharray="3 3" />
              <circle cx={x(active)} cy={y(a.mrr)} r={5} strokeWidth={2}
                      className="fill-white stroke-teal-700 dark:fill-slate-900 dark:stroke-teal-400" />
            </g>
          )}
        </svg>
      )}

      {a && active !== null && (
        <div
          role="status"
          className="pointer-events-none absolute top-2 rounded-lg border border-slate-200 bg-white px-3 py-2 text-sm shadow-md dark:border-slate-700 dark:bg-slate-800"
          style={{ left: Math.min(Math.max(x(active) - 56, 0), Math.max(width - 120, 0)) }}
        >
          <p className="text-slate-600 dark:text-slate-300">{a.month}</p>
          <p className="font-semibold tabular-nums text-slate-900 dark:text-slate-50">{usd(a.mrr)}</p>
        </div>
      )}

      {/* Screen-reader data table */}
      <table className="sr-only">
        <caption>MRR by month</caption>
        <thead><tr><th>Month</th><th>MRR</th></tr></thead>
        <tbody>{data.map((d) => <tr key={d.month}><td>{d.month}</td><td>{usd(d.mrr)}</td></tr>)}</tbody>
      </table>
    </div>
  );
}

export default function Dashboard({
  data, loading = false, onRetry,
}: { data: DashboardData | null; loading?: boolean; onRetry?: () => void }) {
  const { dark, toggle } = useTheme();
  const [q, setQ] = useState("");

  const mrr = data?.mrr ?? [];
  const users = data?.users ?? [];
  const tx = data?.transactions ?? [];

  const kpis = useMemo(() => {
    const cur = mrr.at(-1)?.mrr ?? null;
    const prev = mrr.at(-2)?.mrr ?? null;
    const mrrDelta = cur !== null && prev ? ((cur - prev) / prev) * 100 : null;
    const half = Math.floor(users.length / 2);
    const avg = (a: UsersPoint[]) => a.reduce((s, d) => s + d.dau, 0) / (a.length || 1);
    const recent = users.slice(half), before = users.slice(0, half);
    const usersDelta = before.length && avg(before) ? ((avg(recent) - avg(before)) / avg(before)) * 100 : null;
    return {
      mrr: cur, mrrDelta,
      arr: cur !== null ? cur * 12 : null,
      dau: users.at(-1)?.dau ?? null, usersDelta,
    };
  }, [mrr, users]);

  const filteredTx = useMemo(() => {
    const s = q.trim().toLowerCase();
    return s ? tx.filter((t) => [t.customer, t.email, t.plan].some((v) => v.toLowerCase().includes(s))) : tx;
  }, [tx, q]);

  return (
    <div className="min-h-screen bg-slate-50 text-slate-900 dark:bg-slate-950 dark:text-slate-50">
      <header className="sticky top-0 z-20 border-b border-slate-200 bg-white/90 backdrop-blur dark:border-slate-800 dark:bg-slate-950/90">

                       dark:border-slate-700 dark:text-slate-200 dark:hover:bg-slate-800"

            {dark ? <Sun className="size-5" aria-hidden /> : <Moon className="size-5" aria-hidden />}

        {/* KPIs */}
        <div className="mt-6 grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
          <Kpi label="Monthly recurring revenue" icon={DollarSign} loading={loading}
               value={kpis.mrr !== null ? usd(kpis.mrr) : null} delta={kpis.mrrDelta} />
          <Kpi label="Annualized run rate" icon={Repeat} loading={loading}
               value={kpis.arr !== null ? usd(kpis.arr, true) : null} delta={kpis.mrrDelta} />
          <Kpi label="Daily active users" icon={Users} loading={loading}

          title="Recent transactions" className="mt-4"
          action={
            tx.length > 0 && (
              <label className="relative block w-40 sm:w-64">
                <span className="sr-only">Search transactions</span>
                <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-slate-500" aria-hidden />

                  type="search" value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search"
                  className="min-h-11 w-full rounded-lg border border-slate-300 bg-white py-3 pl-9 pr-4 text-base
                             text-slate-900 placeholder:text-slate-500 focus-visible:outline-2 focus-visible:outline-teal-600

            <TransactionsTable rows={filteredTx} filtered={q.trim() !== ""} onClearFilter={() => setQ("")} />
          )}
        </Card>
      </main>
    </div>
  );
}

## 4. What's covered, against the skill's rules

| Requirement | Implementation |
|---|---|
| **Responsive** | Grids go 1 → 2 → 3 columns for the KPIs, and charts stack until `lg` (3/5 + 2/5 split). The table scrolls horizontally on narrow screens. MRR chart axis labels thin out below 480 px, and the chart resizes with `ResizeObserver`. |
| **Defensive spacing** | Containers use `max-w-7xl mx-auto px-4 sm:px-6 md:px-8 lg:px-12`, with `pt-8 sm:pt-10` below the sticky header and `pb-20 sm:pb-32` at the bottom. |
| **Dark mode** | Class-based, with a no-flash init script, system preference as the default and `localStorage` persistence. The toggle has `aria-pressed`, and every colour has a `dark:` pair. |
| **Empty states** | Each chart has its own empty state, and the table has two (no data, and no search results with "Clear search"). There is also a skeleton loading state. The demo harness switches between data, empty and loading. |
