 *

  id: 'cart_1',
  customerId: 'cus_123',
  paymentMethodId: 'pm_abc',
  currency: 'usd',
  items: [
    { sku: 'SKU-A', quantity: 2, unitPrice: 1500 },
    { sku: 'SKU-B', quantity: 1, unitPrice: 999 },
  ],
  ...overrides,
});

    charge: vi.fn().mockResolvedValue({ id: 'ch_1', status: 'succeeded' }),

      await expect(sut.service.checkout(makeCart())).rejects.toThrow();

      expect(sut.stripe.charge).toHaveBeenCalledTimes(1);
    });
  });

      await expect(sut.service.checkout(makeCart({ items: [] }))).rejects.toThrow();

      expect(sut.inventory.reserve).not.toHaveBeenCalled();
      expect(sut.stripe.charge).not.toHaveBeenCalled();

      await expect(sut.service.checkout(cart)).rejects.toThrow();

      expect(sut.inventory.reserve).not.toHaveBeenCalled();

        .mockResolvedValueOnce({ reservationId: 'res_fail' })
